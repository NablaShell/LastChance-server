// src/network/config.rs
use anyhow::{anyhow, Result};
use serde::{Deserialize, Serialize};
use crate::storage::SafeFSOps;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Config {
    pub server_url: String,
    pub file_server_url: String,
    pub mask_header_name: String,
    pub mask_header_value: String,
}

impl Default for Config {
    fn default() -> Self {
        Self {
            server_url: "https://msg.example.com".to_string(),
            file_server_url: "https://files.example.com".to_string(),
            mask_header_name: "X-DNS-Connect".to_string(),
            mask_header_value: "We_Are_Not_Legion".to_string(),
        }
    }
}

impl Config {
    pub fn load(base_dir: &str) -> Result<Self> {
        let safe_fs = SafeFSOps::new(base_dir)?;
        let config_path = "global.conf";
        
        match safe_fs.safe_read_file(config_path) {
            Ok(data) => {
                let mut cfg: Config = serde_json::from_slice(&data)
                    .map_err(|e| anyhow!("invalid config file: {}", e))?;
                
                // Fill empty fields with defaults
                let defaults = Config::default();
                if cfg.server_url.is_empty() {
                    cfg.server_url = defaults.server_url;
                }
                if cfg.file_server_url.is_empty() {
                    cfg.file_server_url = defaults.file_server_url;
                }
                if cfg.mask_header_name.is_empty() {
                    cfg.mask_header_name = defaults.mask_header_name;
                }
                if cfg.mask_header_value.is_empty() {
                    cfg.mask_header_value = defaults.mask_header_value;
                }
                
                Ok(cfg)
            }
            Err(_) => {
                // File doesn't exist, create with defaults
                let default_cfg = Config::default();
                let json_data = serde_json::to_string_pretty(&default_cfg)
                    .map_err(|e| anyhow!("failed to marshal default config: {}", e))?;
                
                safe_fs.safe_write_file(config_path, json_data.as_bytes(), 0o600)?;
                
                Ok(default_cfg)
            }
        }
    }
}
// src/network/listener.rs
use anyhow::{anyhow, Result};
use ed25519_dalek::{Signer, SigningKey};
use reqwest::header::CONTENT_TYPE;
use reqwest::Client;
use std::time::Duration;
use crate::network::Config;

/// Один из допустимых размеров пакета (см. API.md §4.1).
const ALLOWED_PACKET_SIZES: [usize; 4] = [256, 1024, 4096, 65536];

pub struct Listener {
    client: Client,
    config: Config,
}

impl Listener {
    pub fn new(config: &Config) -> Self {
        let client = Client::builder()
            .timeout(Duration::from_secs(60))
            .pool_max_idle_per_host(10)
            .build()
            .unwrap_or_default();

        Self {
            client,
            config: config.clone(),
        }
    }

    pub async fn pull_messages_with_auth(
        &self,
        identity_hash: &str,
        pub_key_hex: &str,          // hex(ed25519_public_key) — НЕ хеш!
        signing_key: &SigningKey,
    ) -> Result<Vec<Vec<u8>>> {
        let now = chrono::Utc::now().timestamp().to_string();
        // message = timestamp + target_hash (без разделителя, API.md §4.2)
        let message = format!("{}{}", now, identity_hash);
        let signature = signing_key.sign(message.as_bytes());
        let sig_hex = hex::encode(signature.to_bytes());

        let url = format!("{}/pull/{}", self.config.server_url, identity_hash);

        let response = self.client
            .get(&url)
            .header("X-Identity-Key", pub_key_hex)
            .header("X-Timestamp", &now)
            .header("X-Signature", &sig_hex)
            .header("X-Requested-With", "XMLHttpRequest")
            .header("Accept", "multipart/form-data")
            .header("User-Agent", "Mozilla/5.0 (X11; Arch Linux; rv:130.0) Gecko/20100101 Firefox/130.0")
            .header(&self.config.mask_header_name, &self.config.mask_header_value)
            .send()
            .await
            .map_err(|e| anyhow!("pull request failed: {}", e))?;

        let status = response.status();

        if status == reqwest::StatusCode::NO_CONTENT {
            return Ok(Vec::new());
        }

        if status == reqwest::StatusCode::TOO_MANY_REQUESTS {
            return Err(anyhow!("rate limited (429)"));
        }

        if status == reqwest::StatusCode::UNAUTHORIZED {
            let body = response.text().await.unwrap_or_default();
            return Err(anyhow!("unauthorized (401): {}", body));
        }

        if !status.is_success() {
            let body = response.text().await.unwrap_or_default();
            return Err(anyhow!("server returned {}: {}", status, body));
        }

        // Content-Type должен быть multipart/form-data; boundary=...
        let content_type = response
            .headers()
            .get(CONTENT_TYPE)
            .and_then(|v| v.to_str().ok())
            .unwrap_or_default()
            .to_string();

        let bytes = response
            .bytes()
            .await
            .map_err(|e| anyhow!("read body: {}", e))?;

        if !content_type.starts_with("multipart/") {
            // API.md гарантирует multipart на 200, но если сервер вернул
            // что-то иное — не молча делаем вид, что это один пакет.
            return Err(anyhow!(
                "unexpected Content-Type on /pull: {:?}",
                content_type
            ));
        }

        let messages = parse_multipart(&content_type, &bytes)?;

        // Валидация размеров (API.md §4.1: ровно 256/1024/4096/65536)
        for m in &messages {
            if !ALLOWED_PACKET_SIZES.contains(&m.len()) {
                log::warn!(
                    "[LISTENER] packet size {} is not in allowed grid {:?}",
                    m.len(),
                    ALLOWED_PACKET_SIZES
                );
            }
        }

        Ok(messages)
    }

    pub async fn download_file(&self, file_hash: &str) -> Result<Vec<u8>> {
        if file_hash.is_empty() {
            return Err(anyhow!("empty file_hash"));
        }

        let url = format!("{}/download/{}", self.config.file_server_url, file_hash);

        let response = self.client
            .get(&url)
            .header("X-Requested-With", "XMLHttpRequest")
            .header("User-Agent", "Mozilla/5.0 (X11; Arch Linux; rv:130.0) Gecko/20100101 Firefox/130.0")
            .header(&self.config.mask_header_name, &self.config.mask_header_value)
            .send()
            .await
            .map_err(|e| anyhow!("download request failed: {}", e))?;

        match response.status().as_u16() {
            200 => {
                let data = response
                    .bytes()
                    .await
                    .map_err(|e| anyhow!("read file data: {}", e))?;
                Ok(data.to_vec())
            }
            404 => Err(anyhow!(
                "file not found (already downloaded or expired)"
            )),
            400 => Err(anyhow!("bad request (missing file hash)")),
            429 => Err(anyhow!("rate limited (429)")),
            code => {
                let body = response.text().await.unwrap_or_default();
                Err(anyhow!("download failed: {} - {}", code, body))
            }
        }
    }
}

// ---------------------------------------------------------------------------
// Multipart parsing
// ---------------------------------------------------------------------------

/// Извлекает `boundary` из `Content-Type: multipart/form-data; boundary=...`.
fn extract_boundary(content_type: &str) -> Result<Vec<u8>> {
    // boundary может быть в кавычках: boundary="abc"
    for part in content_type.split(';') {
        let part = part.trim();
        if let Some(rest) = part.strip_prefix("boundary=") {
            let b = rest.trim().trim_matches('"');
            if b.is_empty() {
                return Err(anyhow!("empty multipart boundary"));
            }
            return Ok(b.as_bytes().to_vec());
        }
    }
    Err(anyhow!("no boundary in Content-Type: {:?}", content_type))
}

/// Разбирает multipart-тело ответа `/pull`.
///
/// Формат (RFC 7578):
///   --boundary\r\n
///   headers...\r\n
///   \r\n
///   <payload>\r\n
///   --boundary\r\n
///   ...
///   --boundary--\r\n
fn parse_multipart(content_type: &str, body: &[u8]) -> Result<Vec<Vec<u8>>> {
    let boundary = extract_boundary(content_type)?;

    // Разделители: "\r\n--boundary" — между частями,
    //             "--boundary"     — в начале тела.
    let mut delimiter = Vec::with_capacity(boundary.len() + 4);
    delimiter.extend_from_slice(b"\r\n--");
    delimiter.extend_from_slice(&boundary);

    // В начале тела delimiter идёт без ведущего \r\n.
    let mut leading = Vec::with_capacity(boundary.len() + 2);
    leading.extend_from_slice(b"--");
    leading.extend_from_slice(&boundary);

    let mut messages = Vec::new();
    let mut pos;

    // Первый boundary
    if body.starts_with(&leading) {
        pos = leading.len();
    } else if let Some(idx) = find_subsequence(body, &delimiter) {
        pos = idx + delimiter.len();
    } else {
        return Err(anyhow!("multipart body has no boundary"));
    }

    loop {
        // После boundary: либо "--" (конец), либо "\r\n" (заголовки части)
        if body.get(pos..pos + 2) == Some(b"--") {
            break;
        }
        if body.get(pos..pos + 2) != Some(b"\r\n") {
            return Err(anyhow!("malformed multipart: expected CRLF after boundary"));
        }
        pos += 2;

        // Ищем конец заголовков части (пустая строка)
        let headers_end = find_subsequence(&body[pos..], b"\r\n\r\n")
            .ok_or_else(|| anyhow!("multipart part has no header terminator"))?;
        let payload_start = pos + headers_end + 4;

        // Ищем следующий delimiter
        let next = find_subsequence(&body[payload_start..], &delimiter)
            .ok_or_else(|| anyhow!("multipart body ended without closing boundary"))?;

        let payload_end = payload_start + next;
        if payload_end > payload_start {
            messages.push(body[payload_start..payload_end].to_vec());
        }

        pos = payload_end + delimiter.len();
    }

    Ok(messages)
}

fn find_subsequence(haystack: &[u8], needle: &[u8]) -> Option<usize> {
    if needle.is_empty() || haystack.len() < needle.len() {
        return None;
    }
    haystack
        .windows(needle.len())
        .position(|w| w == needle)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn parses_two_parts() {
        let body = b"--XyZ\r\nContent-Type: application/octet-stream\r\n\r\nPART1\r\n--XyZ\r\nContent-Type: application/octet-stream\r\n\r\nPART2\r\n--XyZ--\r\n";
        let ct = "multipart/form-data; boundary=XyZ";
        let parts = parse_multipart(ct, body).unwrap();
        assert_eq!(parts.len(), 2);
        assert_eq!(parts[0], b"PART1");
        assert_eq!(parts[1], b"PART2");
    }

    #[test]
    fn parses_empty_multipart() {
        let body = b"--XyZ--\r\n";
        let ct = "multipart/form-data; boundary=XyZ";
        let parts = parse_multipart(ct, body).unwrap();
        assert!(parts.is_empty());
    }

    #[test]
    fn quoted_boundary() {
        let body = b"--abc\r\n\r\nX\r\n--abc--\r\n";
        let ct = r#"multipart/form-data; boundary="abc""#;
        let parts = parse_multipart(ct, body).unwrap();
        assert_eq!(parts, vec![b"X".to_vec()]);
    }
}

// src/network/sender.rs
use anyhow::{anyhow, Result};
use reqwest::Client;
use std::time::Duration;
use crate::network::Config;

const MAX_UPLOAD_SIZE: usize = 100 * 1024 * 1024; // 100 MB
const ALLOWED_PACKET_SIZES: [usize; 4] = [256, 1024, 4096, 65536];

pub struct Sender {
    client: Client,
    config: Config,
}

impl Sender {
    pub fn new(config: &Config) -> Self {
        // Один клиент на всё; таймаут 120s покрывает upload до 100 MB.
        let client = Client::builder()
            .timeout(Duration::from_secs(120))
            .pool_max_idle_per_host(10)
            .build()
            .unwrap_or_default();

        Self {
            client,
            config: config.clone(),
        }
    }

    pub async fn send_message(&self, target_hash: &str, packet: &[u8]) -> Result<()> {
        if !ALLOWED_PACKET_SIZES.contains(&packet.len()) {
            return Err(anyhow!(
                "invalid packet size {} (allowed: {:?})",
                packet.len(),
                ALLOWED_PACKET_SIZES
            ));
        }

        let url = format!("{}/push/{}", self.config.server_url, target_hash);

        let max_retries = 3;
        let base_delay = Duration::from_secs(1);
        let mut last_err: Option<String> = None;

        for attempt in 0..=max_retries {
            let result = self
                .client
                .post(&url)
                .header("Content-Type", "application/octet-stream")
                .header("Content-Length", packet.len().to_string())
                .header("X-Requested-With", "XMLHttpRequest")
                .header("User-Agent", "Mozilla/5.0 (X11; Ubuntu; Linux x86_64; rv:149.0) Gecko/20100101 Firefox/149.0")
                .header(&self.config.mask_header_name, &self.config.mask_header_value)
                // bytes::Bytes — без лишней копии Vec
                .body(bytes::Bytes::copy_from_slice(packet))
                .send()
                .await;

            match result {
                Ok(resp) => {
                    let status = resp.status();
                    if status.is_success() {
                        return Ok(());
                    }
                    if status.as_u16() == 429 {
                        // rate limit — единственный статус, на который ретраим по API.md
                        let body = resp.text().await.unwrap_or_default();
                        last_err = Some(format!("rate limited: {}", body));
                        if attempt < max_retries {
                            let delay = base_delay * 2u32.pow(attempt as u32);
                            log::info!(
                                "[SENDER] 429, retry in {:?} (attempt {}/{})",
                                delay,
                                attempt + 1,
                                max_retries + 1
                            );
                            tokio::time::sleep(delay).await;
                            continue;
                        }
                    } else {
                        // прочие 4xx/5xx — не ретраим
                        let body = resp.text().await.unwrap_or_default();
                        return Err(anyhow!("server returned {}: {}", status, body));
                    }
                }
                Err(e) => {
                    last_err = Some(format!("request failed: {}", e));
                    if attempt < max_retries {
                        let delay = base_delay * 2u32.pow(attempt as u32);
                        tokio::time::sleep(delay).await;
                        continue;
                    }
                }
            }
        }

        Err(anyhow!(
            "push failed after {} attempts: {}",
            max_retries + 1,
            last_err.unwrap_or_else(|| "unknown".into())
        ))
    }

    pub async fn upload_file(&self, payload: &[u8], file_name: &str) -> Result<String> {
        if payload.is_empty() {
            return Err(anyhow!("empty payload"));
        }
        if payload.len() > MAX_UPLOAD_SIZE {
            return Err(anyhow!(
                "file too large: {} bytes (max 100MB)",
                payload.len()
            ));
        }

        let url = format!("{}/upload", self.config.file_server_url);

        let response = self.client
            .post(&url)
            .header("Content-Type", "application/octet-stream")
            // Content-Length обязателен (API.md §5.1). body(Bytes) его ставит сам,
            // но явный заголовок защищает от регрессий.
            .header("Content-Length", payload.len().to_string())
            .header("X-File-Name", file_name)
            .header("X-Requested-With", "XMLHttpRequest")
            .header("User-Agent", "Mozilla/5.0 (X11; Ubuntu; Linux x86_64; rv:149.0) Gecko/20100101 Firefox/149.0")
            .header(&self.config.mask_header_name, &self.config.mask_header_value)
            .body(bytes::Bytes::copy_from_slice(payload))
            .send()
            .await
            .map_err(|e| anyhow!("upload request failed: {}", e))?;

        match response.status().as_u16() {
            200 => {
                let file_hash = response
                    .text()
                    .await
                    .map_err(|e| anyhow!("read response: {}", e))?;
                let file_hash = file_hash.trim().to_string();
                if file_hash.is_empty() {
                    return Err(anyhow!("server returned empty file hash"));
                }
                Ok(file_hash)
            }
            411 => Err(anyhow!("missing Content-Length")),
            413 => Err(anyhow!("file too large (413)")),
            429 => Err(anyhow!("rate limited (429)")),
            code => {
                let body = response.text().await.unwrap_or_default();
                Err(anyhow!("upload failed: {} - {}", code, body))
            }
        }
    }
}
